8. 组件 Props 接口

export interface TooltipProps {
  content?: string | React.ReactNode;
  variant?: 'dark' | 'glass' | 'light';

  // 地图模式（经纬度定位）
  longitude?: number;
  latitude?: number;

  // DOM 模式（挂载到目标元素）
  targetElement?: HTMLElement | null;

  placement?: 'top' | 'right' | 'bottom' | 'left';
  offset?: number;         // 默认 8px
  trigger?: 'hover' | 'click';
  visible?: boolean;
  title?: string;
  items?: TooltipItem[];
  className?: string;
}

export interface TooltipItem {
  label: string;
  value: string | number;
}
