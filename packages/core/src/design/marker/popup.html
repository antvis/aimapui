6. 组件 Props 接口

export interface PopupProps {
  longitude: number;            // 经度
  latitude: number;             // 纬度
  content?: string | React.ReactNode; // 纯文本 / HTML / ReactNode
  closeButton?: boolean;          // 是否显示关闭按钮
  size?: 'compact' | 'standard' | 'detailed'; // 尺寸变体
  header?: PopupHeader;           // 结构化标题栏
  attributes?: PopupAttribute[];    // 属性列表
  actions?: PopupAction[];        // 底部操作按钮
  visible?: boolean;              // 受控可见性
  onClose?: () => void;            // 关闭回调
  className?: string;             // 自定义类名
}
