# Schema 类型定义

aimapui 提供完整的 TypeScript 类型定义,帮助你编写类型安全的代码。

## AimapSchema

完整的 Schema 类型:

```tsx
import type { AimapSchema } from '@antv/aimapui'

const schema: AimapSchema = {
  map: MapSchema
  layers?: LayerSchema[]
  controls?: ControlSchema[]
  interactions?: InteractionSchema[]
  legend?: LegendSchema
  responsive?: ResponsiveSchema
  events?: EventSchema
}
```

## MapSchema

地图底图配置:

```tsx
import type { MapSchema } from '@antv/aimapui'

const mapSchema: MapSchema = {
  basemap: 'gaode' | 'mapbox' | 'tianditu' | 'tencent' | 'baidu' | 'map'
  token?: string
  style?: MapStylePreset | string
  center?: [number, number]
  zoom?: number
  pitch?: number
  rotation?: number
  minZoom?: number
  maxZoom?: number
  bounds?: [[number, number], [number, number]]
  gestureConfig?: GestureConfig
}

type MapStylePreset = 'light' | 'dark' | 'normal' | 'darkblue' | 'satellite'

interface GestureConfig {
  dragPan?: boolean
  pinchZoom?: boolean
  dragRotate?: boolean
}
```

## LayerSchema

图层配置:

```tsx
import type { LayerSchema, LayerType, SourceType } from '@antv/aimapui'

type LayerType = 'point' | 'line' | 'polygon' | 'heatmap' | 'raster' | 'image'
type SourceType = 'geojson' | 'json' | 'csv'

const layerSchema: LayerSchema = {
  id?: string
  type: LayerType
  name?: string
  visible?: boolean
  zIndex?: number
  minZoom?: number
  maxZoom?: number
  autoFit?: boolean
  
  // 数据源
  source: unknown
  sourceType?: SourceType
  sourceConfig?: SourceConfig
  
  // 视觉映射
  color?: string
  colorField?: string
  colorValues?: string[] | string
  
  size?: number
  sizeField?: string
  sizeValues?: number[]
  
  shape?: string
  shapeField?: string
  shapeValues?: string[]
  
  style?: Record<string, unknown>
  
  // 过滤
  filterField?: string
  filterValues?: unknown[]
  
  // 动画
  animate?: AnimateConfig
  
  // 交互
  active?: ActiveConfig
  select?: SelectConfig
  
  // 信息窗体
  enablePopup?: boolean
  popupFields?: string[]
}
```

## ControlSchema

控件配置:

```tsx
import type { ControlSchema, ControlType, ControlPosition } from '@antv/aimapui'

type ControlType = 
  | 'zoom'
  | 'scale'
  | 'fullscreen'
  | 'geoLocate'
  | 'mapTheme'
  | 'mouseLocation'
  | 'exportImage'
  | 'layerSwitch'

type ControlPosition =
  | 'topleft'
  | 'topright'
  | 'bottomleft'
  | 'bottomright'
  | 'topcenter'
  | 'bottomcenter'

const controlSchema: ControlSchema = {
  type: ControlType
  position?: ControlPosition
  options?: Record<string, unknown>
}
```

## InteractionSchema

交互元素配置:

```tsx
import type { InteractionSchema, MarkerSchema, PopupSchema, TooltipSchema } from '@antv/aimapui'

const marker: MarkerSchema = {
  type: 'marker'
  longitude: number
  latitude: number
  content?: string
  draggable?: boolean
}

const popup: PopupSchema = {
  type: 'popup'
  longitude: number
  latitude: number
  content: string
  closeButton?: boolean
}

const tooltip: TooltipSchema = {
  type: 'tooltip'
  content: string
  trigger?: 'hover' | 'click'
}
```

## LegendSchema

图例配置:

```tsx
import type { 
  LegendSchema,
  LegendCategoriesSchema,
  LegendRampSchema,
  LegendProportionSchema,
  LegendIconSchema
} from '@antv/aimapui'

const categories: LegendCategoriesSchema = {
  type: 'categories'
  title?: string
  labels: string[]
  colors: string[]
}

const ramp: LegendRampSchema = {
  type: 'ramp'
  title?: string
  labels: string[]
  colors: string[]
  isContinuous?: boolean
}

const proportion: LegendProportionSchema = {
  type: 'proportion'
  title?: string
  labels: [number, number][]
  fillColor?: string
}

const icon: LegendIconSchema = {
  type: 'icon'
  title?: string
  items: LegendIconItem[]
}
```

## 完整类型导出

```tsx
import type {
  // 主类型
  AimapSchema,
  
  // 子类型
  MapSchema,
  LayerSchema,
  ControlSchema,
  InteractionSchema,
  LegendSchema,
  ResponsiveSchema,
  EventSchema,
  
  // 枚举类型
  BasemapType,
  LayerType,
  SourceType,
  ControlType,
  ControlPosition,
  MapStylePreset,
  
  // 配置类型
  GestureConfig,
  SourceConfig,
  AnimateConfig,
  ActiveConfig,
  SelectConfig,
  
  // 图例类型
  LegendCategoriesSchema,
  LegendRampSchema,
  LegendProportionSchema,
  LegendIconSchema,
  LegendIconItem
} from '@antv/aimapui'
```
